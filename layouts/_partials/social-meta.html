{{- /* Open Graph and Twitter card tags for link previews (LinkedIn, Slack, X, ...). Hugo ships built-in versions; these differ on purpose: title and description come from head.html, so previews match and the meta description; only blog posts are og:type "article"; every page uses the default preview image. Expects a dict: page, title, description. */ -}} {{- $p := .page }} <meta property="og:site_name" content="{{ site.Title }}"> <meta property="og:title" content="{{ .title }}"> <meta property="og:description" content="{{ .description }}"> <meta property="og:url" content="{{ $p.Permalink }}"> <meta property="og:locale" content="{{ replace site.Language.Locale "-" "_" }}"> {{- if and $p.IsPage (eq $p.Section "blog") }} <meta property="og:type" content="article"> <meta property="article:published_time" content="{{ $p.PublishDate.Format "2006-01-02T15:04:05-07:00" }}"> <meta property="article:modified_time" content="{{ $p.Lastmod.Format "2006-01-02T15:04:05-07:00" }}"> {{- range $p.GetTerms "tags" }} <meta property="article:tag" content="{{ .Data.Term }}"> {{- end }} {{- else }} <meta property="og:type" content="website"> {{- end }} {{- with resources.Get "images/og.png" | fingerprint }} <meta property="og:image" content="{{ .Permalink }}"> <meta property="og:image:width" content="{{ .Width }}"> <meta property="og:image:height" content="{{ .Height }}"> <meta property="og:image:alt" content="{{ site.Title }}: {{ site.Params.description }}"> {{- end }} <meta name="twitter:card" content="summary_large_image">
